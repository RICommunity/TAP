# fschat provides the conversation templates (fastchat.conversation); its base
# install does NOT pull in torch, so it is safe here.
# 0.2.36 relaxes the old "pydantic<2" cap (0.2.32 hard-pinned it), so it now
# coexists with chromadb's pydantic>=2 requirement without a resolver conflict.
fschat==0.2.36
numpy==1.26.3

pytz==2023.3.post1

# protobuf / pydantic are intentionally NOT pinned here: chromadb (below) pulls
